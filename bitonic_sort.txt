uint32_t rot_bits_left_32(uint32_t in, uint32_t s) {

    return (in << s) | (in >> (32 - s)) & 0xFFFF;

}

uint32_t Xoshiro_32(vector<uint32_t>& seed) {
    uint32_t res = rot_bits_left_32(seed[1] * 5, 7) * 9;
    uint32_t t = (seed[1]) << 17;

    seed[2] ^= seed[0];
    seed[3] ^= seed[1];
    seed[1] ^= seed[2];
    seed[0] ^= seed[3];

    seed[2] ^= t;
    seed[3] ^= rot_bits_left_32(seed[3], 45);

    return res;
}

void sort_vals(vector<int>& list) {
    int s = list.size();
    int max_step = 2;
    for (int i = 0; i < s; i++) {
        for (int st = 0; st < s / 2; st++) {
            if (list[st * 2] > list[st * 2 + 1]) {
                int temp_0 = list[st * 2];
                list[st * 2] = list[st * 2 + 1];
                list[st * 2 + 1] = temp_0;
            }
        }

        max_step *= 2;

        for (int k = 0; k < s / 2; k++) {

            if (list[k] > list[k + max_step - 1]) {
                int temp_1 = list[k];
                list[k] = list[k + max_step - 1];
                list[k + max_step - 1] = temp_1;
            }


        }
    }

}



void bitonicSeqMerge(vector<int>& a, int start, int BseqSize, int direction) {

    if (BseqSize > 1) {

        int k = BseqSize / 2;
        for (int i = start; i < start + k; i++) {
            if (direction == (a[i] > a[i + k]))
                swap(a[i], a[i + k]);
        }
        bitonicSeqMerge(a, start, k, direction);
        bitonicSeqMerge(a, start + k, k, direction);


    }
}

void bitonic_sort(vector<int>& a, int start, int BseqSize, int direction) {
    if (BseqSize > 1) {

        int k = BseqSize / 2;

        bitonic_sort(a, start, k, 1);
        bitonic_sort(a, start + k, k, 0);

        bitonicSeqMerge(a, start, BseqSize, direction);
        if (direction == 1 && start == 0) {
            cout << " start = " << start << " BseqSize = " << BseqSize << " direction = " << direction << '\n';
            output_to_str(a);
        }

    }
}

void ascendingSwap(int index1, int index2, vector<int>& ar) //swap two values such that they appear in ascending order in the array	
{
    if (ar[index2] < ar[index1])
    {
        int temp = ar[index2];
        ar[index2] = ar[index1];
        ar[index1] = temp;
    }
}

void descendingSwap(int index1, int index2, vector<int>& ar) //swap two values such that they appear in ascending order in the array	
{
    if (ar[index2] > ar[index1])
    {
        int temp = ar[index2];
        ar[index2] = ar[index1];
        ar[index1] = temp;
    }
}

void sort_bitonic(int start, int end, vector<int>& arr) {

    int size_arr = arr.size();
    //the size of the blocks gets bigger by factor of 2 every iteration
    for (int k = 2; k < size_arr; k *= 2) {
        //jump/iterate by the size of a block, send the rescpective indeces to the below method
        for (int i = 0; i < size_arr; i = i + k) {

            //estimate which block we are at to know if it should ascend or descend	
            if (((i / k) % 2) == 0) {
                bitonic_sort_one_dir(i, i + k - 1, 1, arr);
            }
            else {
                bitonic_sort_one_dir(i, i + k - 1, 0, arr);
            }
            cout << "ARRAYS AT STAGES OF SORT BITONIC : \n";
            output_to_str(arr);

        }

    }

}
